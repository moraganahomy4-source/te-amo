import turtle import math screen = turtle.Screen() screen.bgcolor("black") t = turtle.Turtle() t.hideturtle() t.color("#ff66c1") for i in range(120): angle = i * (math.pi * 2) / 120 x = 16 * (math.sin(angle) ** 3) y = (13 * math.cos(angle) - 5 * math.cos(2 * angle) - 2 * math.cos(3 * angle) - math.cos(4 * angle)) t.penup() t.goto(x * 15, y * 15) t.write("TE AMO", align="center", font=("Arial", 7, "bold")) # TE AMO en el centro t.goto(0, -20) t.write("TE AMO", align="center", font=("Arial", 18, "bold")) turtle.done()
